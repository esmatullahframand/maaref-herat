// ============================================ // بارگذاری اطلاعات کارمندان و معاشات // ============================================ async function loadData() { // بررسی اعتبار توکن if (!token || token === 'undefined' || token === 'null') { showToast('نشست کاربری معتبر نیست. دوباره وارد شوید.', true); return; } // دریافت اطلاعات از API async function fetchApi(url) { const response = await fetch(url, { method: 'GET', headers: { 'Authorization': 'Bearer ' + token, 'Accept': 'application/json' } }); const raw = await response.text(); console.log('درخواست API:', url); console.log('وضعیت HTTP:', response.status); console.log('پاسخ سرور:', raw); let result; try { result = raw ? JSON.parse(raw) : null; } catch (error) { throw new Error( url + ' پاسخ JSON معتبر ارسال نکرد. وضعیت: ' + response.status ); } if (!response.ok) { throw new Error( url + ': ' + ( result?.error || result?.message || response.statusText || 'خطای سرور' ) + ' (HTTP ' + response.status + ')' ); } if (result === null) { throw new Error(url + ': پاسخ سرور خالی است.'); } return result; } // استخراج آرایه از پاسخ API function extractArray(result, keys, apiName) { if (Array.isArray(result)) { return result; } if (result && typeof result === 'object') { for (const key of keys) { if (Array.isArray(result[key])) { return result[key]; } } } throw new Error( apiName + ': ساختار اطلاعات دریافتی معتبر نیست.' ); } try { // ---------------------------------------- // ۱. دریافت فهرست کارمندان // ---------------------------------------- const recordsResponse = await fetchApi('/api/records'); cachedRecords = extractArray( recordsResponse, ['records', 'data', 'employees'], 'فهرست کارمندان' ); console.log( 'تعداد کارمندان دریافت‌شده:', cachedRecords.length ); // ---------------------------------------- // ۲. آماده‌سازی فهرست انتخاب کارمند // ---------------------------------------- const employeeSelect = document.getElementById( 'newSalaryEmployee' ); if (!employeeSelect) { throw new Error( 'عنصر انتخاب کارمند با شناسه newSalaryEmployee پیدا نشد.' ); } employeeSelect.innerHTML = ''; cachedRecords.forEach(record => { let info = record.data || {}; // اگر اطلاعات کارمند به شکل رشته JSON باشد if (typeof info === 'string') { try { info = JSON.parse(info); } catch (error) { console.error( 'خطا در خواندن اطلاعات کارمند:', record.id, error ); info = {}; } } const firstName = info.firstName || info.firstname || ''; const lastName = info.lastName || info.lastname || ''; const jobTitle = info.jobTitle || info.job_title || ''; const option = document.createElement('option'); option.value = record.id; option.textContent = `${firstName} ${lastName} (${jobTitle})`; employeeSelect.appendChild(option); }); // ---------------------------------------- // ۳. دریافت اطلاعات معاشات // ---------------------------------------- const salariesResponse = await fetchApi('/api/salaries'); cachedSalaries = extractArray( salariesResponse, ['salaries', 'data', 'records'], 'اطلاعات معاشات' ); console.log( 'تعداد معاشات دریافت‌شده:', cachedSalaries.length ); // ---------------------------------------- // ۴. بررسی اطلاعات دریافتی // ---------------------------------------- cachedSalaries = cachedSalaries.filter(salary => { if (!salary || typeof salary !== 'object') { console.warn( 'رکورد نامعتبر معاش حذف شد:', salary ); return false; } return true; }); // ---------------------------------------- // ۵. محاسبه آمار // ---------------------------------------- updateStats(); // ---------------------------------------- // ۶. نمایش فهرست معاشات // ---------------------------------------- renderTreeMenu(cachedSalaries); console.log( 'بارگذاری اطلاعات معاشات با موفقیت انجام شد.' ); } catch (error) { console.error( 'خطا در بارگذاری اطلاعات معاشات:', error ); showToast( error.message || 'خطا در بارگذاری اطلاعات', true ); } } // ============================================ // محاسبه آمار معاشات // ============================================ function updateStats() { const salaries = Array.isArray(cachedSalaries) ? cachedSalaries : []; let total = salaries.length; let entitlement = 0; let deductions = 0; let netPay = 0; salaries.forEach(salary => { entitlement += Number(salary.total_entitlement) || 0; deductions += Number(salary.total_deductions) || 0; netPay += Number(salary.net_pay) || 0; }); // تعداد کل ثبت‌شده‌ها document.getElementById('statTotal').textContent = total.toLocaleString(); // مجموع استحقاق document.getElementById('statEnt').textContent = entitlement.toLocaleString(); // مجموع کسرات document.getElementById('statDed').textContent = deductions.toLocaleString(); // مبلغ قابل پرداخت document.getElementById('statNet').textContent = netPay.toLocaleString(); }
